45void UOB_HotbarWidget::SetSelectedSlot(int32 InSelectedSlotIndex) {
46 if (!InventoryComp) return;
47
48 int32 TargetIndex = (InSelectedSlotIndex > 0) ? (InSelectedSlotIndex - 1) : INDEX_NONE;
49
50 if (TargetIndex != INDEX_NONE && TargetIndex == SelectedSlotIndex) {
51 TargetIndex = INDEX_NONE;
52 InventoryComp->RemoveFromHand();
53 }
54
55 if (Slots.IsValidIndex(SelectedSlotIndex)) {
56 Slots[SelectedSlotIndex]->PlaySelectAnim(true);
57 }
58
59 SelectedSlotIndex = TargetIndex;
60
61 if (Slots.IsValidIndex(SelectedSlotIndex)) {
62 Slots[SelectedSlotIndex]->PlaySelectAnim(false);
63 }
64
65 InventoryComp->TakeToHand(SelectedSlotIndex);
66
67 UE_LOG(LogTemp, Warning, TEXT("Current Selected Slot: %d"), SelectedSlotIndex);
