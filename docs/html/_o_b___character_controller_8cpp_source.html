12void AOB_CharacterController::BeginPlay()
13{
14 Super::BeginPlay();
15
16 if (!IsLocalController()) return;
17
18 if (ULocalPlayer* LocalPlayer = GetLocalPlayer())
19 {
20 if (UEnhancedInputLocalPlayerSubsystem* EnhancedSubsystem = ULocalPlayer::GetSubsystem<UEnhancedInputLocalPlayerSubsystem>(LocalPlayer))
21 {
22 if (DefaultMappingContext)
23 {
24 EnhancedSubsystem->AddMappingContext(DefaultMappingContext, 0);
25 }
26 }
27 }
28
29 if (HudWidgetClass && IsLocalController())
30 {
31 PlayerHudWidget = CreateWidget<UOB_PlayerHud>(Cast<APlayerController>(this), HudWidgetClass);
32 if (PlayerHudWidget) {
33 PlayerHudWidget->AddToPlayerScreen();
34
35 if (AOB_BaseCharacter* BaseChar = GetPawn<AOB_BaseCharacter>())
36 {
37 ControlledCharacter = BaseChar;
38 PlayerHudWidget->Init(ControlledCharacter);
39 }
40 }
41 }
42}

166void AOB_CharacterController::SelectHotbarSlot(const FInputActionValue& Value) {
167 if (bBlockMovement) return;
168
169 if (!ControlledCharacter) return;
170
171 int32 SlotIndex = FMath::RoundToInt(Value.Get<float>());
172
173 auto Inventory = IOB_CharacterInterface::Execute_GetInventoryComp(ControlledCharacter);
174 if (Inventory)
175 Inventory->OnSelectedSlotChanged.Broadcast(SlotIndex);
176}
