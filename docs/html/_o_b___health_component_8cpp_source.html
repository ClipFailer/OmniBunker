5float UOB_HealthComponent::ApplyDamage(const float Damage) {
6 if (Damage <= 0)
7 return CurrentHealth;

9 CurrentHealth = FMath::Max(CurrentHealth - Damage, 0);

11 OnDamagedDelegate.Broadcast(CurrentHealth);

14 Die();
