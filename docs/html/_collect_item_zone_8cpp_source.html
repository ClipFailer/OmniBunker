69bool ACollectItemZone::AddItem(UOB_InventoryItemData* ItemData) {
70 if (!ItemData) return false;
71
72 // Finding the existing item to increase total amount, else creating new ItemCollection.
73 if (FItemCollection* ItemCollection = CollectedItems.FindByPredicate(
74 [ItemData](const FItemCollection& ItemCollection) {
75 return ItemCollection.ItemData == ItemData;
76 }
77 )
78 ) {
79 ItemCollection->TotalAmount += ItemData->MaxStackSize;
80
81 UE_LOG(
82 LogTemp,
83 Display,
84 TEXT("ACollectItemZone::AddItem: Item %s added. Total amount: %d"),
85 *ItemData->ItemName.ToString(),
86 ItemCollection->TotalAmount
87 );
88
89 } else {
90 FItemCollection NewItemCollection;
91 NewItemCollection.ItemData = ItemData;
92 NewItemCollection.TotalAmount = ItemData->MaxStackSize;
93
94 CollectedItems.Add(NewItemCollection);
95
96 UE_LOG(
97 LogTemp,
98 Display,
99 TEXT("ACollectItemZone::AddItem: Item %s added. Total amount: %d"),
100 *ItemData->ItemName.ToString(),
101 NewItemCollection.TotalAmount
102 );
103 }
104
105 return true;
106}
