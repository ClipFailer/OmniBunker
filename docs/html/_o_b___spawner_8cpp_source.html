56TSubclassOf<AOB_InteractableActor> AOB_Spawner::GetRandomItemClass() const {
57 if (PossibleItems.IsEmpty()) return nullptr;
58
59 float TotalWeight = 0.f;
60 for (const auto& ItemClass : PossibleItems) {
61 if (!IsValid(ItemClass)) continue;
62
63 const AOB_InteractableActor* DefaultItem = ItemClass->GetDefaultObject<AOB_InteractableActor>();
64 if (DefaultItem) {
65 TotalWeight += DefaultItem->GetSpawnWeight();
66 }
67 }
68
69 if (TotalWeight <= 0.f) return nullptr;
70
71 float RandomWeight = FMath::FRandRange(0.f, TotalWeight);
72 for (const auto& ItemClass : PossibleItems) {
73 if (!ItemClass) continue;
74
75 const AOB_InteractableActor* DefaultItem = ItemClass->GetDefaultObject<AOB_InteractableActor>();
76 if (!DefaultItem) continue;
77
78 const float ItemWeight = DefaultItem->GetSpawnWeight();
79
80 if (RandomWeight <= ItemWeight) {
81 return ItemClass;
82 }
83
84 RandomWeight -= ItemWeight;
85 }
86
87 return PossibleItems[0];
88}
