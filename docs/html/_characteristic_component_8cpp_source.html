36void UCharacteristicComponent::GenerateRandomCharacteristics() {
37 Characteristics.Empty();
38
39 for (const auto& Pool : CharacteristicsPool) {
40 if (!IsValid(Pool)) {
41 UE_LOG(
42 LogTemp,
43 Warning,
44 TEXT("CharacteristicsPool contains an invalid or null Pool entry!")
45 );
46 continue;
47 }
48
49 TSubclassOf<UOB_CharacterCharacteristic> CharClass = Pool->GetRandomCharacteristic();
50 if (!IsValid(CharClass)) {
51 UE_LOG(LogTemp, Warning, TEXT("Characteristic Class is null!"));
52 continue;
53 }
54
55 UOB_CharacterCharacteristic* Char = NewObject<UOB_CharacterCharacteristic>(this, CharClass);
56 if (IsValid(Char)) {
57 UE_LOG(
58 LogTemp,
59 Warning,
60 TEXT("Characteristic: %s has loaded."),
61 *Char->GetDisplayName().ToString()
62 );
63 Characteristics.Add(Char);
