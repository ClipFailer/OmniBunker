50void UStaminaComponent::SpendStamina(const float Value) {
51 if (Value <= 0) return;
52
53 CurrentStamina = FMath::Max(CurrentStamina - Value, 0);
54
55 OnStaminaDecreasedDelegate.Broadcast(CurrentStamina);
56
57 UE_LOG(LogTemp, Warning, TEXT("Spended stamina: %f(-%f)"), CurrentStamina, Value);
58}

72void UStaminaComponent::IncreaseStamina() {
73 CurrentStamina = FMath::Min(CurrentStamina + IncreaseStaminaValue, MaxStamina);
74
75 OnStaminaIncreasedDelegate.Broadcast(CurrentStamina);
76
77 if (CurrentStamina >= MaxStamina) {
78 if (GetWorld()) {
79 if (GetWorld()->GetTimerManager().IsTimerActive(IncreaseStaminaTimer))
80 GetWorld()->GetTimerManager().ClearTimer(IncreaseStaminaTimer);
81 }
82 }
83
84 UE_LOG(LogTemp, Warning, TEXT("Increased stamina: %f"), CurrentStamina);
85}
